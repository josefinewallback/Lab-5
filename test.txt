CXX = g++
CXXFLAGS = -std=c++17 -O2 -Wall -Wextra 

# -Wall -Wextra : Show all and extra warnings

SRC_DIR = src
TARGET = slm

SRCS_FILES = $(SRC_DIR)/main.cpp $(SRC_DIR)/kgram.cpp $(SRC_DIR)/smallmodel.cpp
OBJS = $(SRCS_FILES:.cpp=.o) # kgram.o, smallmodel.o


$(TARGET): $(OBJS)
	$(CXX) $(CXXFLAGS) $^ -o $@

$(SRC_DIR)/%.o: $(SRC_DIR)/%.cpp
	$(CXX) $(CXXFLAGS) -c $< -o $@

clean:
	rm -f $(SRC_DIR)/*.o $(TARGET)

.PHONY: all clean
